name: Daily internship scan

on:
  schedule:
    # 9am Eastern = 13:00 UTC (EDT) or 14:00 UTC (EST). Both run; the script only emails
    # once per day and only after 9am ET, and never skips a posting.
    - cron: "5 13 * * *"
    - cron: "5 14 * * *"
  workflow_dispatch:
    inputs:
      force_email:
        description: "Send the email now even if already sent today"
        type: boolean
        default: false

permissions:
  contents: write

concurrency:
  group: tracker
  cancel-in-progress: false

jobs:
  scan:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
          cache: pip
      - run: pip install -r requirements.txt
      - name: Scan and email
        env:
          GMAIL_USER: ${{ secrets.GMAIL_USER }}
          GMAIL_APP_PASSWORD: ${{ secrets.GMAIL_APP_PASSWORD }}
          TO_EMAIL: ${{ secrets.TO_EMAIL }}
          USAJOBS_API_KEY: ${{ secrets.USAJOBS_API_KEY }}
          USAJOBS_EMAIL: ${{ secrets.USAJOBS_EMAIL }}
          ADZUNA_APP_ID: ${{ secrets.ADZUNA_APP_ID }}
          ADZUNA_APP_KEY: ${{ secrets.ADZUNA_APP_KEY }}
          PAGE_URL: https://${{ github.repository_owner }}.github.io/${{ github.event.repository.name }}/
        run: python tracker.py ${{ inputs.force_email && '--force-email' || '' }}
      - name: Save results
        run: |
          git config user.name "internship-bot"
          git config user.email "bot@users.noreply.github.com"
          git add state.json index.html
          git diff --cached --quiet || git commit -m "Daily scan $(date -u +%F)"
          git push
